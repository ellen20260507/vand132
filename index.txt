// 构造函数：初始化exe同目录路径（核心修改）
MyRequestHandler::MyRequestHandler(QObject* parent)
    : stefanfrings::HttpRequestHandler(parent)
{
    // 获取exe所在目录
    QString appDir = QCoreApplication::applicationDirPath();
    // 拼接exe同目录下的文件路径
    yellowTxtPath = appDir + "/yellow.txt";
    redTxtPath = appDir + "/red.txt";
    backgroundImagePath = appDir + "/background.jpg"; // 背景图同目录

    // 打印路径日志，方便调试
    qDebug() << "MyRequestHandler初始化完成，文件路径：";
    qDebug() << "yellow.txt：" << yellowTxtPath;
    qDebug() << "red.txt：" << redTxtPath;
    qDebug() << "背景图：" << backgroundImagePath;
    qDebug() << "黄点数据从 TXT 文件读取";
}

                // 1. 系统时间更新（不变）
                function updateSystemTime() {
                    const now = new Date();
                    const timeStr = now.toLocaleString("zh-CN", {
                        year: "numeric", month: "2-digit", day: "2-digit",
                        hour: "2-digit", minute: "2-digit", second: "2-digit"

                // 2. 核心：缓存 TXT 中的「设备标识→状态」映射（W1→1，T2→2 等）
                let deviceStatusMap = {}; // key: 设备标识（W1/T2），value: 状态（1/2/3）
                function fetchDeviceStatus() {
                    fetch('/detectedpoints').then(res => res.json()).then(points => {
                        points.forEach(p => {
                            deviceStatusMap[p.address] = p.status;
                        });
                        console.log("黄点状态缓存：", deviceStatusMap); // 新增打印
                    });
                    fetch('/detectedred').then(res => res.json()).then(points => {
                        points.forEach(p => {
                            deviceStatusMap[p.address] = p.status;
                        });
                        console.log("红点状态缓存：", deviceStatusMap); // 新增打印
                    });
                }
                // 初始化缓存
                fetchDeviceStatus();
                setInterval(fetchDeviceStatus, 3000); // 和TXT刷新频率一致

                // 3. 串口数据渲染（核心修改：关联设备标识+状态+十进制转换）
                let lastSerialData = {};
                function getDeviceIdFromSerialKey(serialKey) {
                    if (serialKey.length !== 8) return "未知设备"; // 串口key格式：4位地址+4位功能码
                    const addrHex = serialKey.slice(0, 4); // 前4位：地址（十六进制）
                    const funcCode = serialKey.slice(4, 8); // 后4位：功能码
                    // 功能码→设备前缀（W/T/E）
                    let prefix = "?";
                    if (funcCode === "1011") prefix = "W";   // 腕带→W
                    else if (funcCode === "1014") prefix = "T"; // 台垫→T
                    else if (funcCode === "1021") prefix = "E"; // 设备→E
                    // 十六进制地址→十进制（如"0001"→1）
                    const addrDec = parseInt(addrHex, 16);
                    return `${prefix}${addrDec}`; // 最终设备标识：W1/T2/E3
                }
                // 辅助函数：十六进制数组→十进制数组
                function hexToDecArray(hexArray) {
                    return hexArray.map(hex => {
                        const dec = parseInt(hex, 16);
                        return isNaN(dec) ? "无效" : dec.toString();
                    });
                }
                // 辅助函数：状态码→文字+颜色（1=正常绿，2=警告黄，3=异常红）
                function getStatusTextAndColor(statusCode) {
                    switch(statusCode) {
                        case 1: return { text: "正常", color: "#48bb78" };
                        case 2: return { text: "请检查连接或佩戴状况", color: "#ffc107" };
                        case 3: return { text: "异常", color: "#999999" };
                        default: return { text: "未知", color: "#666" };
                    }
                }
                                        function fetchSerialData() {
                                            // 1. 获取数据容器（确保始终能访问，避免报错）
                                            const container = document.getElementById("dataContainer");
                                            if (!container) {
                                                console.error("未找到数据容器元素，无法渲染串口数据");
                                                return;
                                            }

                                            // 2. 发起请求获取串口数据
                                            fetch('/getdata')
                                                .then(response => {
                                                    // 检查HTTP响应状态（非200均视为失败）
                                                    if (!response.ok) {
                                                        throw new Error(`服务器返回异常：${response.status}（${response.statusText}）`);
                                                    }
                                                    return response.json(); // 解析JSON数据
                                                })
                                                .then(serialData => {
                                                    // 3. 调试日志：打印原始串口数据（方便排查后端是否返回有效数据）
                                                    console.log("【串口数据】原始数据：", serialData);

                                                    // 4. 对比上一次数据，避免重复渲染（减少无效刷新）
                                                    const currentDataStr = JSON.stringify(serialData);
                                                    if (currentDataStr === JSON.stringify(lastSerialData)) {
                                                        console.log("【串口数据】与上一次相同，跳过渲染");
                                                        return;
                                                    }
                                                    lastSerialData = serialData; // 更新缓存

                                                    // 5. 处理空数据（后端无有效数据时）
                                                    if (Object.keys(serialData).length === 0) {
                                                        container.innerHTML = '<div class="no-data">暂无串口数据（设备未响应或未配置）</div>';
                                                        return;
                                                    }

                                                    // 6. 渲染有效数据（核心逻辑）
                                                    let html = "";
                                                    let renderIndex = 0; // 用于动画延迟，避免同时渲染卡顿

                                                    // 遍历所有串口数据（key：地址+功能码，value：十六进制数据数组）
                                                    for (const [serialKey, hexArray] of Object.entries(serialData)) {
                                                        // 6.1 解析设备标识（如从"00011011"得到"W1"）
                                                        const deviceId = getDeviceIdFromSerialKey(serialKey);
                                                        console.log(`【串口数据】解析标识：serialKey=${serialKey} → deviceId=${deviceId}`);

                                                        // 6.2 获取设备状态（从TXT缓存中匹配，无匹配则显示“待同步”）
                                                        const statusCode = deviceStatusMap[deviceId];
                                                        let statusInfo;
                                                        if (statusCode === undefined) {
                                                            // 未匹配到状态（标识不一致或TXT未加载）
                                                            statusInfo = { text: "待同步", color: "#666" };
                                                            console.warn(`【状态匹配】未找到标识${deviceId}的状态（检查TXT文件和串口解析是否一致）`);
                                                        } else {
                                                            // 匹配到状态，转换为文字和颜色
                                                            statusInfo = getStatusTextAndColor(statusCode);
                                                        }

                                                        // 6.3 转换十六进制数据为十进制（处理空/无效数据）
                                                        const decArray = hexToDecArray(hexArray);
                                                        // 判断是否全为无效数据（避免显示0）
                                                        const hasValidData = decArray.some(item => item !== "无效" && item !== "无数据");
                                                        const displayData = hasValidData ? decArray.join(" , ") : "无有效数据";

                                                        // 6.4 生成HTML（带动画延迟，提升视觉体验）
                                                        html += `<div class="data-item" style="animation-delay: ${renderIndex * 0.05}s">
                                                            <div><strong>设备标识：</strong>${deviceId}</div>
                                                            <div><strong>状态：</strong><span style="color: ${statusInfo.color}">${statusInfo.text}</span></div>
                                                            <div><strong>数据：</strong>${displayData}</div>
                                                        </div>`;

                                                        renderIndex++;
                                                    }

                                                    // 7. 渲染到页面
                                                    container.innerHTML = html;
                                                })
                                                .catch((error) => {
                                                    // 8. 错误处理（网络失败、JSON解析错误等）
                                                    console.error("【串口数据获取失败】详细原因：", error);
                                                    container.innerHTML = `<div class="no-data" style="color: #f53f3f;">
                                                        数据获取失败：${error.message}<br>
                                                        请检查后端服务或网络连接
                                                    </div>`;
                                                });
                                        }
                // 初始化渲染+定时刷新
                fetchSerialData();
                setInterval(fetchSerialData, 3000);

                // 4. 黄点/红点渲染+弹窗逻辑（不变，保留原有）
                let imgRenderInfo = { width: 0, height: 0 };

                function fetchDetectedYellowPoints() {

                            else if (point.status === 2) el.classList.add("point-color-yellow");

                            });
                            container.appendChild(el);
                        });
                    });
                }

                fetchDetectedRedPoints();
                setInterval(fetchDetectedYellowPoints, 2000);
                setInterval(fetchDetectedRedPoints, 2000);
        </script>
    </body>
    </html>
            )html";
            response.setHeader("Content-Type", "text/html; charset=utf-8");
            response.write(html.toUtf8(), true);
            return;
        }

    // 2. 背景图接口：/showimage（替换为exe同目录路径）
    else if (method == "GET" && path == "/showimage") {
        QFile imageFile(backgroundImagePath); // 用初始化的同目录路径

        // 检查文件是否存在
        if (!imageFile.exists()) {
            qDebug() << "背景图不存在：" << backgroundImagePath;
            response.setStatus(404, "Image Not Found");
            response.write(QString("背景图不存在（路径：%1）").arg(backgroundImagePath).toUtf8(), true);
            return;
        }

        // 检查文件是否能打开
        if (!imageFile.open(QIODevice::ReadOnly)) {
            qDebug() << "打开背景图失败：" << imageFile.errorString();
            response.setStatus(500, "Internal Error");
            response.write(QString("打开背景图失败（原因：%1）").arg(imageFile.errorString()).toUtf8(), true);
            return;
        }

        // 读取图片数据并返回
        QByteArray imageData = imageFile.readAll();
        imageFile.close();
        response.setHeader("Content-Type", "image/jpeg");
        response.setHeader("Content-Length", QString::number(imageData.size()).toUtf8());
        response.write(imageData, true);
        return;
    }

    // 3. 黄点数据接口：/detectedpoints（替换为exe同目录路径）
    else if (method == "GET" && path == "/detectedpoints") {
        QFile txtFile(yellowTxtPath); // 用初始化的同目录路径

        if (!txtFile.exists()) {
            qDebug() << "设备TXT文件不存在：" << yellowTxtPath;
            response.setStatus(404, "File Not Found");
            response.write("设备检测数据文件不存在", true);
            return;
        }

        if (!txtFile.open(QIODevice::ReadOnly | QIODevice::Text)) {
            qDebug() << "打开设备TXT文件失败：" << txtFile.errorString();
            response.setStatus(500, "File Open Failed");
            response.write(QString("打开设备数据文件失败（原因：%1）").arg(txtFile.errorString()).toUtf8(), true);
            return;
        }

        QJsonArray pointArray;
        QTextStream in(&txtFile);
        while (!in.atEnd()) {
            QString line = in.readLine().trimmed();
            if (line.isEmpty()) continue;

            // 解析“x,y|地址|状态”格式
            QStringList lineParts = line.split('|');
            if (lineParts.size() != 3) {
                qDebug() << "无效黄点格式（跳过）：" << line;
                continue;
            }

            // 解析坐标（x,y）
            QStringList coordParts = lineParts[0].split(',');
            if (coordParts.size() != 2) {
                qDebug() << "无效坐标格式（跳过）：" << lineParts[0];
                continue;
            }

            // 转换坐标为浮点数
            bool xOk = false, yOk = false;
            float x = coordParts[0].toFloat(&xOk);
            float y = coordParts[1].toFloat(&yOk);
            if (xOk && yOk) {
                bool statusOk = false;
                int status = lineParts[2].toInt(&statusOk);
                if (!statusOk || (status != 1 && status != 2 && status != 3)) {
                    qDebug() << "黄点状态值无效（默认2=黄）：" << lineParts[2];
                    status = 2;
                }
                QJsonObject pointObj;
                pointObj["x"] = x;
                pointObj["y"] = y;
                pointObj["address"] = lineParts[1]; // 设备标识（W1/T2/E3）
                pointObj["status"] = status;
                pointArray.append(pointObj);
            } else {
                qDebug() << "坐标转换失败（跳过）：" << line;
            }
        }
        txtFile.close();

        response.setHeader("Content-Type", "application/json; charset=utf-8");
        response.write(QJsonDocument(pointArray).toJson(QJsonDocument::Compact), true);
        return;
    }

    // 4. 串口数据接口：/getdata（无路径修改，保留原有逻辑）
    else if (method == "GET" && path == "/getdata") {
        MainWindow* mainWindow = qobject_cast<MainWindow*>(this->parent());
        if (!mainWindow) {
            qDebug() << "获取MainWindow实例失败（无法读取串口数据）";
            response.setStatus(500, "Internal Error");
            response.write("无法获取串口数据（MainWindow实例异常）", true);
            return;
        }

        QMap<QString, QList<QString>> serialData = mainWindow->getAddressFuncData();
        QJsonObject dataObj;

        for (auto it = serialData.begin(); it != serialData.end(); ++it) {
            QJsonArray dataArray;
            for (const QString& dataItem : it.value()) {
                dataArray.append(dataItem);
            }
            dataObj[it.key()] = dataArray;
        }

        response.setHeader("Content-Type", "application/json; charset=utf-8");
        response.write(QJsonDocument(dataObj).toJson(QJsonDocument::Compact), true);
        return;
    }

    // 5. 红点数据接口：/detectedred（替换为exe同目录路径）
    else if (method == "GET" && path == "/detectedred") {
        QFile txtFile(redTxtPath); // 用初始化的同目录路径

        if (!txtFile.exists()) {
            qDebug() << "红点TXT文件不存在：" << redTxtPath;
            response.setStatus(404, "File Not Found");
            response.write("红点检测数据文件不存在", true);
            return;
        }

        if (!txtFile.open(QIODevice::ReadOnly | QIODevice::Text)) {
            qDebug() << "打开红点TXT文件失败：" << txtFile.errorString();
            response.setStatus(500, "File Open Failed");
            response.write(QString("打开红点数据文件失败（原因：%1）").arg(txtFile.errorString()).toUtf8(), true);
            return;
        }

        QJsonArray pointArray;
        QTextStream in(&txtFile);
        while (!in.atEnd()) {
            QString line = in.readLine().trimmed();
            if (line.isEmpty()) continue;

            // 解析“x,y|地址|状态”格式
            QStringList lineParts = line.split('|');
            if (lineParts.size() != 3) {
                qDebug() << "无效红点格式（跳过）：" << line;
                continue;
            }

            QStringList coordParts = lineParts[0].split(',');
            if (coordParts.size() != 2) {
                qDebug() << "无效红点坐标格式（跳过）：" << lineParts[0];
                continue;
            }

            bool xOk = false, yOk = false;
            float x = coordParts[0].toFloat(&xOk);
            float y = coordParts[1].toFloat(&yOk);
            if (xOk && yOk) {
                bool statusOk = false;
                int status = lineParts[2].toInt(&statusOk);
                if (!statusOk || (status != 1 && status != 2 && status != 3)) {
                    qDebug() << "红点状态值无效（默认2=黄）：" << lineParts[2];
                    status = 2;
                }
                QJsonObject pointObj;
                pointObj["x"] = x;
                pointObj["y"] = y;
                pointObj["address"] = lineParts[1]; // 设备标识（E1/E2）
                pointObj["status"] = status;
                pointArray.append(pointObj);
            } else {
                qDebug() << "红点坐标转换失败（跳过）：" << line;
            }
        }
        txtFile.close();

        response.setHeader("Content-Type", "application/json; charset=utf-8");
        response.write(QJsonDocument(pointArray).toJson(QJsonDocument::Compact), true);
        return;
    }

    // 6. 其他路径：返回404
    else {
        response.setStatus(404, "Not Found");
        response.write(QString("请求路径不存在：%1").arg(path).toUtf8(), true);
    }
}
